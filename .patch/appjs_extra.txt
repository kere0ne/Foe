async function loadConversations(){try{const list=await api('/api/conversations');const box=$('#chatList');if(!box)return;box.innerHTML='';if(!list.length){box.className='sidebar-empty';box.textContent='Past chats show up here.';return}box.className='projects';list.forEach(c=>{const b=document.createElement('button');b.className='project';const label=document.createElement('span');label.textContent=c.title||'Chat';label.title=c.title||'Chat';const del=document.createElement('small');del.textContent='×';del.title='Delete this chat';del.onclick=async e=>{e.stopPropagation();try{await api('/api/conversations/'+c.id,{method:'DELETE'});if(conversationId===c.id)conversationId=null;loadConversations()}catch(e){toast(e.message)}};b.append(label,del);b.onclick=()=>openConversation(c.id);box.append(b)})}catch(e){}}
async function openConversation(id){try{const d=await api('/api/conversations/'+id+'/messages');conversationId=id;messages.length=0;$('#messages').innerHTML='';const hero=document.querySelector('.hero');if(hero)hero.classList.add('hidden');d.messages.forEach(m=>{addMessage(m.role,m.content);messages.push({role:m.role,content:m.content})});$('#heading').textContent='Past chat';$('#sidebar').classList.remove('mobile-open');window.scrollTo(0,0)}catch(e){toast(e.message)}}
